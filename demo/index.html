A stage in a few lines

import { Stage, Layer, Rect, Transformer } from "vizat";

const stage = new Stage({ container: "board", width: 800, height: 500 });
const layer = new Layer();
stage.add(layer);

const card = new Rect({
  x: 80, y: 80, width: 180, height: 110,
  fill: "#243044", stroke: "#e8a25a", cornerRadius: 16,
  draggable: true,
});
layer.add(card, new Transformer({ nodes: [card] }));
